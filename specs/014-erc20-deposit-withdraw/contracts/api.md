## Part 1: Smart Contract API (Existing)

### `executeAction` — ZK-Verified Pool Investment

The existing `executeAction` function handles privacy-preserving investments.

```solidity
/// @notice Execute an action using a ZK proof (Spend + Change flow)
/// @param proof ZK proof data generated by Noir circuit
/// @param root Merkle tree root at time of proof generation
/// @param nullifierHash Public nullifier hash (prevents double-spending)
/// @param changeCommitment Commitment to the change note (UTXO)
/// @param actionHash Hash of Uniswap v4 parameters (anti-tampering)
/// @param investAmount Amount being invested
/// @param uniswapParams Encoded Uniswap v4 parameters (reserved for future use)
function executeAction(
    bytes calldata proof,
    bytes32 root,
    bytes32 nullifierHash,
    bytes32 changeCommitment,
    bytes32 actionHash,
    uint256 investAmount,
    bytes calldata uniswapParams
) external nonReentrant;
```

**Location**: `PrivacyVault.sol` lines 280-365

**Preconditions**:
- ZK proof verifies successfully
- `nullifiers[nullifierHash] == false`
- `root == currentRoot`
- `actionHash != bytes32(0)`
- `changeCommitment != bytes32(0)`

**Effects**:
- `nullifiers[nullifierHash] = true`
- `commitments[changeCommitment] = true`
- `currentRoot = keccak256(currentRoot, changeCommitment)`
- `nextLeafIndex++`

**Events emitted**:
- `ActionExecuted(nullifierHash, changeCommitment, actionHash, investAmount, timestamp, changeIndex)`
- `MerkleRootUpdated(oldRoot, newRoot, leafCount)`

**Gas target**: < 300,000 gas

### Utility: `computeActionHash`

**Location**: `ghostroute-ui/src/utils/utxo/actionHash.ts` (new file)

```typescript
import { keccak256, encodePacked, toHex } from 'viem';

/**
 * Compute actionHash for pool investment (must match contract implementation)
 * @param poolId - Uniswap v4 pool identifier
 * @param tickLower - Lower tick of LP position
 * @param tickUpper - Upper tick of LP position
 * @param amount0Desired - Token0 amount
 * @param amount1Desired - Token1 amount
 * @returns actionHash as hex string
 */
export function computeActionHash(
  poolId: `0x${string}`,
  tickLower: number,
  tickUpper: number,
  amount0Desired: bigint,
  amount1Desired: bigint
): `0x${string}` {
  return keccak256(
    encodePacked(
      ['bytes32', 'int24', 'int24', 'uint256', 'uint256'],
      [poolId, tickLower, tickUpper, amount0Desired, amount1Desired]
    )
  );
}
```

### Utility: `isPoolCompatible`

**Location**: `ghostroute-ui/src/utils/utxo/poolCompatibility.ts` (new file)

```typescript
import { EnrichedPool } from '@/components/UniswapPoolsSection';
import { Note } from '@/types/utxo/note';
import { getWETHAddress } from '@/config/tokens';

/**
 * Check if a note's token is compatible with a pool
 * @param pool - Target Uniswap v4 pool
 * @param note - User's deposited note
 * @returns true if note token matches one of pool's tokens
 */
export function isPoolCompatible(pool: EnrichedPool, note: Note): boolean {
  const noteToken = note.token.toLowerCase();
  const isETH = noteToken === '0x0000000000000000000000000000000000000000';
  const weth = getWETHAddress(pool.chainId)?.toLowerCase();
  
  const poolToken0 = pool.token0.id.toLowerCase();
  const poolToken1 = pool.token1.id.toLowerCase();
  
  // ETH notes can invest in pools with WETH
  if (isETH && weth) {
    return poolToken0 === weth || poolToken1 === weth;
  }
  
  // ERC20 notes must match token0 or token1
  return poolToken0 === noteToken || poolToken1 === noteToken;
}

/**
 * Get the investment side for a note in a pool
 * @returns 0 if note matches token0, 1 if token1, -1 if not compatible
 */
export function getInvestmentSide(pool: EnrichedPool, note: Note): 0 | 1 | -1 {
  const noteToken = note.token.toLowerCase();
  const isETH = noteToken === '0x0000000000000000000000000000000000000000';
  const weth = getWETHAddress(pool.chainId)?.toLowerCase();
  
  const poolToken0 = pool.token0.id.toLowerCase();
  const poolToken1 = pool.token1.id.toLowerCase();
  
  if (isETH && weth) {
    if (poolToken0 === weth) return 0;
    if (poolToken1 === weth) return 1;
    return -1;
  }
  
  if (poolToken0 === noteToken) return 0;
  if (poolToken1 === noteToken) return 1;
  return -1;
}
```

**Implementation**:
```typescript
export async function handleActionExecuted(
  payload: ActionExecutedPayload
): Promise<{ success: boolean; root?: string }> {
  const { chainId, vaultAddress, changeCommitment, changeIndex, blockNumber } = payload;
  
  // Get or create vault
  const vault = await vaultService.getOrCreateVault(chainId, vaultAddress);
  const tree = vaultService.getTree(vault.id);
  
  // Create leaf hash and insert
  const leafHash = createLeafHash(changeCommitment, changeIndex);
  tree.insert(changeIndex, leafHash);
  
  // Update vault root in database
  await vaultService.updateVaultRoot(chainId, vaultAddress, tree.root, blockNumber);
  
  console.log(`[handleActionExecuted] Added change commitment at index ${changeIndex}`);
  
  return { success: true, root: tree.root };
}
```

### Listener Adapter Extension

**Location**: `ghostroute-zk-api/supabase/functions/_shared/adapters/listener-adapter.ts`

**New Event Signature**:
```solidity
ActionExecuted(bytes32 indexed nullifierHash, bytes32 changeCommitment, bytes32 actionHash, uint256 investAmount, uint256 timestamp, uint256 changeIndex)
```

**Event Topic**:
```typescript
const ACTION_EXECUTED_TOPIC = keccak256(
  toBytes('ActionExecuted(bytes32,bytes32,bytes32,uint256,uint256,uint256)')
);
```

**Adapter Function**:
```typescript
export function adaptActionExecutedEvent(
  event: ListenerEventPayload
): ActionExecutedPayload | null {
  // Validate event signature
  if (event.topics[0] !== ACTION_EXECUTED_TOPIC) {
    return null;
  }
  
  // Decode indexed parameter (nullifierHash)
  const nullifierHash = event.topics[1];
  
  // Decode non-indexed parameters
  const decoded = decodeAbiParameters(
    [
      { name: 'changeCommitment', type: 'bytes32' },
      { name: 'actionHash', type: 'bytes32' },
      { name: 'investAmount', type: 'uint256' },
      { name: 'timestamp', type: 'uint256' },
      { name: 'changeIndex', type: 'uint256' },
    ],
    event.data
  );
  
  return {
    chainId: getChainIdFromAddress(event.address),
    vaultAddress: event.address,
    nullifierHash,
    changeCommitment: decoded[0],
    actionHash: decoded[1],
    investAmount: decoded[2].toString(),
    timestamp: Number(decoded[3]),
    changeIndex: Number(decoded[4]),
    blockNumber: event.blockNumber,
    transactionHash: event.transactionHash,
  };
}
```

## Error Codes

### Frontend Errors

| Error | Message | Cause |
|-------|---------|-------|
| `WALLET_NOT_CONNECTED` | "Connect wallet to invest" | No wallet connected |
| `NO_NOTES_AVAILABLE` | "No unspent notes available" | User has no notes |
| `INCOMPATIBLE_POOL` | "Note token not compatible with pool" | Token mismatch |
| `INSUFFICIENT_FUNDS` | "Insufficient funds in note" | Amount > note value |
| `INVALID_TICK_RANGE` | "Invalid tick range" | tickLower >= tickUpper |
| `PROOF_GENERATION_FAILED` | "Failed to generate proof" | ZK proof error |
| `MERKLE_ROOT_MISMATCH` | "Merkle root outdated" | Tree changed during tx |

### Contract Errors

| Error | Selector | Cause |
|-------|----------|-------|
| `ZK proof verification failed` | N/A (string) | Invalid proof |
| `Nullifier already spent` | N/A (string) | Double spend |
| `Invalid Merkle root` | N/A (string) | Root mismatch |
| `Invalid action hash` | N/A (string) | actionHash == 0 |
| `Invalid change commitment` | N/A (string) | changeCommitment == 0 |
